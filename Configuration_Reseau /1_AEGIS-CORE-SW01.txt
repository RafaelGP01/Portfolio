! ================================================================
! AEGIS-CORE-SW01 - Cisco 3650 - Coeur L3 (routage inter-VLAN, ACL)
! A coller apres "enable", bloc par bloc.
! Regle : un "!" n'est un commentaire QU'EN DEBUT DE LIGNE.
! Change les mots de passe d'exemple.
! Adapte le hostname si ton coeur s'appelle autrement.
! ================================================================
!
! ===== ETAPE 1 : base, VLAN, trunks, SVI =====
configure terminal
hostname AEGIS-CORE-SW01
no ip domain-lookup
ip domain-name aegis-capital.local
service password-encryption
service timestamps log datetime msec
enable secret Aeg1s#En@ble26
username admin-net privilege 15 secret Aeg1s#Adm1n26
banner motd #Acces reserve au personnel autorise d'Aegis Capital. Connexions journalisees.#
!
vtp mode transparent
spanning-tree mode rapid-pvst
!
vlan 10
 name ADMINISTRATION
vlan 20
 name SERVEURS
vlan 30
 name POSTES
vlan 40
 name DMZ
vlan 99
 name GESTION-RESEAU
vlan 999
 name PARKING
 exit
!
! Le coeur est racine STP de tous les VLAN
spanning-tree vlan 10,20,30,40,99 priority 4096
!
ip routing
!
! --- Trunks vers les switchs d'acces ---
! Si "switchport trunk encapsulation dot1q" est refuse, ignore l'erreur.
interface GigabitEthernet1/0/1
 description Trunk vers AEGIS-SW01 Gi0/1 (Administration)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,99
 switchport nonegotiate
 no shutdown
 exit
interface GigabitEthernet1/0/2
 description Trunk vers AEGIS-SW02 Gi0/1 (Serveurs)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 20,99
 switchport nonegotiate
 no shutdown
 exit
interface GigabitEthernet1/0/3
 description Trunk vers AEGIS-SW03 Gi0/1 (DMZ)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 40,99
 switchport nonegotiate
 no shutdown
 exit
!
! --- Point d'acces Wi-Fi (VLAN Postes) ---
interface GigabitEthernet1/0/4
 description AEGIS-AP01 Wi-Fi VLAN Postes
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
!
! --- Lien route vers AEGIS-RT01 ---
interface GigabitEthernet1/0/24
 description Transit vers AEGIS-RT01 Gi0/2
 no switchport
 ip address 10.0.0.1 255.255.255.252
 no shutdown
 exit
!
! --- Ports inutilises (parking + shutdown) ---
interface range GigabitEthernet1/0/5 - 23
 description INUTILISE
 switchport mode access
 switchport access vlan 999
 shutdown
 exit
! Ports uplink du module (si refuse sur ton modele, ignore)
interface range GigabitEthernet1/1/1 - 4
 description INUTILISE
 switchport mode access
 switchport access vlan 999
 shutdown
 exit
!
! --- Passerelles des VLAN (SVI) ---
interface Vlan1
 shutdown
 exit
interface Vlan10
 description GW ADMINISTRATION
 ip address 172.16.10.254 255.255.255.0
 no shutdown
 exit
interface Vlan20
 description GW SERVEURS
 ip address 172.16.20.254 255.255.255.0
 no shutdown
 exit
interface Vlan30
 description GW POSTES
 ip address 172.16.30.254 255.255.255.0
 ip helper-address 172.16.20.10
 no shutdown
 exit
interface Vlan40
 description GW DMZ
 ip address 172.16.40.254 255.255.255.0
 no shutdown
 exit
interface Vlan99
 description GW GESTION-RESEAU
 ip address 172.16.99.254 255.255.255.0
 no shutdown
 exit
!
ip route 0.0.0.0 0.0.0.0 10.0.0.2
!
! --- SSH v2, acces admin limite au VLAN 10 ---
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
access-list 10 permit 172.16.10.0 0.0.0.255
line console 0
 login local
 exec-timeout 10 0
 logging synchronous
 exit
line vty 0 15
 login local
 transport input ssh
 access-class 10 in
 exec-timeout 10 0
 exit
!
! --- Journaux, heure, supervision ---
logging host 172.16.10.10
logging trap informational
ntp server 172.16.20.10
snmp-server community Aeg1sRO RO
end
write memory
!
! ===== ETAPE 2 : ACL (a appliquer APRES avoir valide les pings) =====
configure terminal
!
ip access-list extended ACL_POSTES_IN
! DHCP relaye vers DC01
 permit udp any any eq 67
! Ping de la passerelle
 permit icmp 172.16.30.0 0.0.0.255 host 172.16.30.254
! AD / DNS / Kerberos / NTP / GPO vers DC01
 permit udp 172.16.30.0 0.0.0.255 host 172.16.20.10 eq 53
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.20.10 eq 53
 permit udp 172.16.30.0 0.0.0.255 host 172.16.20.10 eq 123
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.20.10 eq 88
 permit udp 172.16.30.0 0.0.0.255 host 172.16.20.10 eq 88
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.20.10 eq 135
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.20.10 eq 389
 permit udp 172.16.30.0 0.0.0.255 host 172.16.20.10 eq 389
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.20.10 eq 445
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.20.10 eq 464
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.20.10 eq 636
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.20.10 eq 3268
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.20.10 range 49152 65535
! Partages de fichiers
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.20.11 eq 445
! CRM et GLPI en HTTPS
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.20.12 eq 443
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.20.14 eq 443
! Agents Wazuh vers SIEM01
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.10.10 eq 1514
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.10.10 eq 1515
! Site vitrine / espace investisseur (DMZ)
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.40.10 eq 443
! Ping de diagnostic vers les serveurs
 permit icmp 172.16.30.0 0.0.0.255 172.16.20.0 0.0.0.255 echo
! Tout le reste de l'interne est interdit (admin, gestion, transit)
 deny ip 172.16.30.0 0.0.0.255 172.16.0.0 0.0.255.255
 deny ip 172.16.30.0 0.0.0.255 10.0.0.0 0.0.0.3
! Internet
 permit tcp 172.16.30.0 0.0.0.255 any eq 80
 permit tcp 172.16.30.0 0.0.0.255 any eq 443
 permit icmp 172.16.30.0 0.0.0.255 any echo
 deny ip any any
 exit
!
ip access-list extended ACL_DMZ_IN
! Reponses aux sessions ouvertes depuis l'interne ou Internet
 permit tcp 172.16.40.0 0.0.0.255 any established
 permit icmp 172.16.40.0 0.0.0.255 any echo-reply
 permit udp 172.16.40.0 0.0.0.255 eq 161 host 172.16.10.10
 permit icmp 172.16.40.0 0.0.0.255 host 172.16.40.254
! Seul APP01 parle a la base
 permit tcp host 172.16.40.11 host 172.16.20.13 eq 3306
! Journaux vers SIEM01
 permit tcp 172.16.40.0 0.0.0.255 host 172.16.10.10 eq 1514
 permit tcp 172.16.40.0 0.0.0.255 host 172.16.10.10 eq 1515
 permit udp 172.16.40.0 0.0.0.255 host 172.16.10.10 eq 514
! Rien d'autre vers l'interne
 deny ip 172.16.40.0 0.0.0.255 172.16.0.0 0.0.255.255
 deny ip 172.16.40.0 0.0.0.255 10.0.0.0 0.0.0.3
! Internet : DNS, NTP, mises a jour, SMTP pour APP01
 permit udp 172.16.40.0 0.0.0.255 any eq 53
 permit tcp 172.16.40.0 0.0.0.255 any eq 53
 permit udp 172.16.40.0 0.0.0.255 any eq 123
 permit tcp 172.16.40.0 0.0.0.255 any eq 80
 permit tcp 172.16.40.0 0.0.0.255 any eq 443
 permit tcp host 172.16.40.11 any eq 587
 deny ip any any
 exit
!
interface Vlan30
 ip access-group ACL_POSTES_IN in
 exit
interface Vlan40
 ip access-group ACL_DMZ_IN in
 exit
end
write memory
